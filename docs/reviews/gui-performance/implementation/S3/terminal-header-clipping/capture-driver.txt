use datum_gui_protocol::{DockTab, TerminalTabState};
use datum_gui_render::visual_capture::OffscreenRenderer;
fn main() -> anyhow::Result<()> {
    let mut state = datum_gui_protocol::load_fixture_workspace_state();
    state.ui.active_dock_tab = Some(DockTab::Terminal);
    state.ui.dock_height_px = 260;
    state.ui.terminal.status = "close terminal? Enter confirms; Escape cancels".into();
    state.ui.terminal.tabs = (1..=3).map(|i| TerminalTabState {
        session_id: format!("terminal-{i}"), previous_session_id: None,
        label: format!("shell {i}"), event_log_path: String::new(),
        activity_event_count: 0, activity_summary: Vec::new(), active: i == 1,
        attached: true, status: "running".into(), restart_count: 0,
        unread_output: false, unread_bell_count: 0,
    }).collect();
    for width in [1280, 480] {
        let mut renderer = OffscreenRenderer::new(width, 800)?;
        renderer.render_workspace(&state, None)?.save(format!("/tmp/pm045-header-{width}.png"))?;
    }
    Ok(())
}
